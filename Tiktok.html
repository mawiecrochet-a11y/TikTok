  {q:"How much time do you spend on TikTok per day?", o:["Under 2 hours","2–3 hours","3 hours or more"], pts:[0,1,2],

   pct:[52.4,4.8,42.9],

   line:["of surveyed students use it under 2 hours a day.","of surveyed students (1 of 21) fell in the 2–3 hour range.","of surveyed students spend 3 or more hours a day."]},

  {q:"When it's time to study, do you consciously stop scrolling?", o:["Yes, I stop","No, I keep scrolling"], pts:[0,2],

   pct:[71.4,28.6],

   line:["of surveyed students consciously stop scrolling to study, so you're in the majority.","of surveyed students did not report stopping, so you share this with a smaller group."]},

  {q:"Do you feel envious of people or lifestyles you see on TikTok?", o:["No, not really","Yes, sometimes or often"], pts:[0,2],

   pct:[42.9,57.1],

   line:["of surveyed students did not report envy, so you're in the smaller group.","of surveyed students reported feeling envious, so you're in the majority."]}

];

let step = 0, ans = [];

function renderQuiz(){

  const box = $("#quizBox");

  if (step < Q.length){

    const q = Q[step];

    box.innerHTML = `<div class="fade">

      <div class="flex justify-between text-xs text-zinc-500 mb-3"><span>Question ${step+1} of {Math.round(step/Q.length*100)}%</span></div>

      <div class="bar mb-6"><i class="p" style="width:${step/Q.length*100}%"></i></div>

      <h3 class="text-xl font-bold mb-5">${q.q}</h3>

      <div class="space-y-3">${q.o.map((o,i)=>`<button class="opt" data-i="${i}">${o}</button>`).join("")}</div></div>`;

  } else {

    const total = ans.reduce((a,v,i)=>a+Q[i].pts[v],0);

    const res = total <= 1 ? ["Intentional scroller","You keep TikTok in its place, much like the students who use it as a creative tool without letting it take over.","#00f2fe"]

              : total <= 3 ? ["Balancing act","You're in the middle: getting real value from the app while sometimes losing time or feeling its pull.","#ffffff"]

              :              ["Deep in the feed","Your habits match the students most affected by distraction and comparison. Try setting a purpose and a time limit before opening the app.","#fe0979"];

    box.innerHTML = `<div class="fade text-center">

      <p class="text-xs uppercase tracking-widest text-zinc-500">Your result · score ${total} / 6</p>

      <h3 class="text-3xl font-extrabold mt-2" style="color:{res[0]}</h3>

      <p class="text-zinc-400 mt-3 max-w-md mx-auto">${res[1]}</p>

      <div class="text-left mt-8 space-y-3">${Q.map((q,i)=>

        `<div class="rounded-xl bg-white/5 p-4"><p class="text-xs text-zinc-500">${q.q}</p>

         <p class="text-sm mt-1"><b class="text-white">You:</b> ${q.o[ans[i]]}</p>

         <p class="text-sm text-zinc-400 mt-1"><b class="text-[#00f2fe]">${q.pct[ans[i]]}%</b> ${q.line[ans[i]]}</p>

         <div class="bar mt-2"><i class="p" data-w="${q.pct[ans[i]]}"></i></div></div>`).join("")}</div>

      <p class="text-xs text-zinc-500 mt-4">Compared with N = 21 first-year Multimedia Arts respondents. Percentages for the opposite answers are the remainder after the reported figure.</p>

      <button id="retry" class="mt-6 px-6 py-3 rounded-full bg-[#fe0979] font-semibold hover:shadow-[0_0_22px_#fe0979] transition">Retake quiz</button></div>`;

    fillBars(box);

  }

}

$("#quizBox").addEventListener("click", e => {

  if (e.target.closest("#retry")){ step = 0; ans = []; renderQuiz(); }

});

const io = new IntersectionObserver(es => es.forEach(e => {

  if (!e.isIntersecting) return;

  e.target.classList.add("in");

  fillBars(e.target);

  io.unobserve(e.target);

}), {threshold:.12});

document.querySelectorAll(".reveal").forEach(el => io.observe(el));

const reduce = window.matchMedia("(prefers-reduced-motion: reduce)").matches;

document.querySelectorAll("[data-count]").forEach(el => {

  if (reduce) return;

  const end = parseFloat(el.dataset.count), dec = +el.dataset.dec, t0 = performance.now(), dur = 1600;

  (function tick(t){

    const p = Math.min((t - t0)/dur, 1), eased = 1 - Math.pow(1 - p, 3);

    el.textContent = (end * eased).toFixed(dec);

    if (p < 1) requestAnimationFrame(tick); else el.textContent = end.toFixed(dec);

  })(t0);

});

</script>
